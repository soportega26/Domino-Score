import 'package:flutter/material.dart';

class ScoreCard extends StatelessWidget {
  final String teamName;
  final int currentScore;
  final int maxScore;
  final List<int> history;
  final bool isCompact;

  const ScoreCard({
    super.key,
    required this.teamName,
    required this.currentScore,
    required this.maxScore,
    required this.history,
    this.isCompact = false,
  });

  @override
  Widget build(BuildContext context) {
    double progress = maxScore > 0 ? currentScore / maxScore : 0.0;
    if (progress > 1.0) progress = 1.0;

    final bool isWinning = currentScore > 0;

    return Container(
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(20),
        boxShadow: [
          BoxShadow(
            blurRadius: 12,
            offset: const Offset(0, 6),
            color: Colors.black.withValues(alpha: 0.22),
          ),
        ],
      ),
      child: Card(
        elevation: 0,
        margin: EdgeInsets.zero,
        color: Colors.white.withValues(alpha: 0.94),
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
        ),
        child: Padding(
          padding: EdgeInsets.all(isCompact ? 8.0 : 12.0),
          child: Column(
            children: [
              Container(
                width: isCompact ? 36 : 44,
                height: isCompact ? 36 : 44,
                decoration: BoxDecoration(
                  color: const Color(0xFF5D3218).withValues(alpha: 0.10),
                  shape: BoxShape.circle,
                ),
                child: Icon(
                  isWinning ? Icons.groups : Icons.person_outline,
                  color: const Color(0xFF5D3218),
                  size: isCompact ? 20 : 24,
                ),
              ),
              const SizedBox(height: 6),
              SizedBox(
                height: isCompact ? 32 : 38,
                child: Center(
                  child: FittedBox(
                    fit: BoxFit.scaleDown,
                    child: Text(
                      teamName.isEmpty ? 'Jugador' : teamName,
                      maxLines: 1,
                      style: TextStyle(
                        fontSize: isCompact ? 14 : 17,
                        fontWeight: FontWeight.w800,
                        color: const Color(0xFF432612),
                      ),
                    ),
                  ),
                ),
              ),
              FittedBox(
                fit: BoxFit.scaleDown,
                child: Text(
                  '$currentScore',
                  style: TextStyle(
                    fontSize: isCompact ? 38 : 48,
                    height: 1,
                    fontWeight: FontWeight.w900,
                    color: const Color(0xFF5D3218),
                  ),
                ),
              ),
              const SizedBox(height: 4),
              Text(
                'de $maxScore pts',
                style: TextStyle(
                  fontSize: isCompact ? 10 : 12,
                  color: Colors.brown.shade400,
                  fontWeight: FontWeight.w600,
                ),
              ),
              const SizedBox(height: 8),
              ClipRRect(
                borderRadius: BorderRadius.circular(20),
                child: LinearProgressIndicator(
                  value: progress,
                  minHeight: isCompact ? 6 : 8,
                  backgroundColor: Colors.brown.shade100,
                  valueColor: const AlwaysStoppedAnimation<Color>(
                    Color(0xFFD29A42),
                  ),
                ),
              ),
              const SizedBox(height: 8),
              Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(
                    Icons.history,
                    size: isCompact ? 12 : 14,
                    color: Colors.brown.shade400,
                  ),
                  const SizedBox(width: 4),
                  Text(
                    'HISTORIAL',
                    style: TextStyle(
                      fontSize: isCompact ? 9 : 11,
                      fontWeight: FontWeight.w800,
                      letterSpacing: 0.8,
                      color: Colors.brown.shade400,
                    ),
                  ),
                ],
              ),
              const SizedBox(height: 6),
              Expanded(
                child: history.isEmpty
                    ? Center(
                        child: Text(
                          'Sin notas',
                          style: TextStyle(
                            fontSize: isCompact ? 10 : 11,
                            color: Colors.brown.shade300,
                          ),
                        ),
                      )
                    : ListView.builder(
                        padding: EdgeInsets.zero,
                        itemCount: history.length,
                        itemBuilder: (context, index) {
                          final points = history[index];
                          return Container(
                            margin: const EdgeInsets.only(bottom: 4),
                            padding: EdgeInsets.symmetric(
                              horizontal: isCompact ? 6 : 8,
                              vertical: isCompact ? 3 : 5,
                            ),
                            decoration: BoxDecoration(
                              color: const Color(0xFFF4E9D8),
                              borderRadius: BorderRadius.circular(8),
                            ),
                            child: Row(
                              mainAxisAlignment: MainAxisAlignment.spaceBetween,
                              children: [
                                Text(
                                  'M${index + 1}',
                                  style: TextStyle(
                                    fontSize: isCompact ? 9 : 10,
                                    color: Colors.brown.shade500,
                                    fontWeight: FontWeight.w600,
                                  ),
                                ),
                                Text(
                                  '+$points',
                                  style: TextStyle(
                                    fontSize: isCompact ? 11 : 13,
                                    fontWeight: FontWeight.w900,
                                    color: const Color(0xFF5D3218),
                                  ),
                                ),
                              ],
                            ),
                          );
                        },
                      ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
