/// Registro persistente de una partida ganada.
class ScoreWinnerRecord {
  final String mesaId;
  final String winnerName;
  final int points;
  final DateTime date;

  ScoreWinnerRecord({
    required this.mesaId,
    required this.winnerName,
    required this.points,
    required this.date,
  });

  Map<String, dynamic> toJson() {
    return {
      'mesaId': mesaId,
      'winnerName': winnerName,
      'points': points,
      'date': date.toIso8601String(),
    };
  }

  factory ScoreWinnerRecord.fromJson(Map<String, dynamic> json) {
    return ScoreWinnerRecord(
      mesaId: json['mesaId']?.toString() ?? 'Mesa 1',
      winnerName: json['winnerName']?.toString() ?? 'Ganador',
      points: json['points'] is int
          ? json['points'] as int
          : int.tryParse(json['points']?.toString() ?? '') ?? 0,
      date: DateTime.tryParse(
            json['date']?.toString() ?? '',
          ) ??
          DateTime.now(),
    );
  }
}

class ScoreScreen extends StatefulWidget {
  final MatchModel match;
  final VoidCallback onGameFinished;

  const ScoreScreen({
    super.key,
    required this.match,
    required this.onGameFinished,
  });

  @override
  State<ScoreScreen> createState() => _ScoreScreenState();
}

class _ScoreScreenState extends State<ScoreScreen> with WidgetsBindingObserver {
  String selectedBackground = 'dominicano';

  List<ScoreWinnerRecord> _historialGanadores = [];

  bool _isGameOver = false;

  InterstitialAd? _interstitialAd;

  DateTime? _gameStartTime;

  static const String _prefHistorialKey = 'domino_winners_history';
  static const int _maxHistorialRegistros = 10;

   // Clave para guardar la partida en curso



// 1. Guardar la mesa activa en SharedPreferences
Future<void> _guardarMesaActivaAuto() async {
 final String prefKey = 'domino_mesa_activa_${widget.match.id}';

  if (widget.match.estaFinalizado) {
    await _limpiarMesaActivaGuardada();
    return;
  }
  try {
    final prefs = await SharedPreferences.getInstance();
    final String jsonMesa = jsonEncode(widget.match.toJson());
    
    await prefs.setString(prefKey, jsonMesa);
    await prefs.reload();
  } catch (e) {
    debugPrint('Error al guardar la partida activa: $e');
  }
}

Future<void> _limpiarMesaActivaGuardada() async {
  final String prefKey = 'domino_mesa_activa_${widget.match.id}';
  final prefs = await SharedPreferences.getInstance();
  await prefs.remove(prefKey);
}

  static const Map<String, String> backgrounds = {
    'wood': 'Madera clásica',
    'green': 'Mesa verde',
    'blue': 'Mesa azul',
    'dark': 'Oscuro',
    'dominicano': 'Dominicana',
    'tropical': 'Tropical',
  };

  static const Map<String, String> backgroundImages = {
    'wood': 'assets/images/backgrounds/wood.jpg',
    'green': 'assets/images/backgrounds/green.jpg',
    'blue': 'assets/images/backgrounds/blue.jpg',
    'dark': 'assets/images/backgrounds/dark.jpg',
    'dominicano': 'assets/images/backgrounds/dominicano.jpg',
    'tropical': 'assets/images/backgrounds/tropical.jpg',
  };

  void _loadInterstitialAd() {
  InterstitialAd.load(
    adUnitId: 'ca-app-pub-5052339394513121/8727589056',
    request: const AdRequest(),
    adLoadCallback: InterstitialAdLoadCallback(
      onAdLoaded: (ad) {
        
        _interstitialAd = ad;
      },
      onAdFailedToLoad: (error) {
      debugPrint('ERROR INTERSTITIAL: $error');
        _interstitialAd = null;
      },
    ),
  );
}

  void _showInterstitialIfNeeded() {
  if (_gameStartTime == null) return;

  final duration =
      DateTime.now().difference(_gameStartTime!);

  if (duration.inMinutes < 30) {
    debugPrint(
      'Partida menor a 30 minutos. No se muestra anuncio.',
    );
    return;
  }

  if (_interstitialAd != null) {
    

    _interstitialAd!.show();

    _interstitialAd = null;

    _loadInterstitialAd();
  }
}
  @override
  void initState() {
    super.initState();

    _gameStartTime = DateTime.now();

  _loadInterstitialAd();

    WidgetsBinding.instance.addObserver(this);

    _cargarFondo();
    _cargarHistorialPersistente();

    // Evaluar si la partida actual (en curso) ya ha sido terminada
    _evaluarEstadoJuego();

    }

  @override
  void dispose() {
    // 🔹 2. Remover observador para liberar memoria
    WidgetsBinding.instance.removeObserver(this);

    _interstitialAd?.dispose();

    super.dispose();
    
  }

  // 🔹 3. DETECTAR CUANDO LA APP SE MINIMIZA O SE CIERRA
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.paused || state == AppLifecycleState.inactive) {
      // Se ejecuta inmediatamente cuando el usuario minimiza, swipea o cambia de app
      _guardarMesaActivaAuto();
    }
  }

  void _evaluarEstadoJuego() {
    final metaAlcanzada = widget.match.scores.any((s) => s >= widget.match.maxPoints);
    setState(() {
      _isGameOver = widget.match.hasWinner || metaAlcanzada;
    });
  }

  // ============================================================
  // HISTORIAL PERSISTENTE
  // ============================================================

  Future<void> _cargarHistorialPersistente() async {
    final prefs = await SharedPreferences.getInstance();

    final String? historialJson = prefs.getString(_prefHistorialKey);

    if (historialJson == null || historialJson.isEmpty) {
      return;
    }

    try {
      final List<dynamic> decodedList = jsonDecode(historialJson);

      final cargados = decodedList
          .whereType<Map<String, dynamic>>()
          .map(
            (item) => ScoreWinnerRecord.fromJson(item),
          )
          .toList();

      if (!mounted) return;

      setState(() {
        _historialGanadores = cargados;
      });
    } catch (e) {
      debugPrint(
        'Error al cargar historial: $e',
      );
    }
  }

  Future<void> _guardarNuevoGanador(
    ScoreWinnerRecord record,
  ) async {
    final prefs = await SharedPreferences.getInstance();

    _historialGanadores.insert(0, record);

    if (_historialGanadores.length > _maxHistorialRegistros) {
      _historialGanadores = _historialGanadores.sublist(
        0,
        _maxHistorialRegistros,
      );
    }

    final String encoded = jsonEncode(
      _historialGanadores.map((e) => e.toJson()).toList(),
    );

    await prefs.setString(
      _prefHistorialKey,
      encoded,
    );
  }
 
  // ============================================================
  // FONDO
  // ============================================================

  Future<void> _cargarFondo() async {
    final prefs = await SharedPreferences.getInstance();

    if (!mounted) return;

    setState(() {
      selectedBackground =
          prefs.getString('domino_background') ?? 'dominicano';
    });
  }

  Future<void> _guardarBackground(
    String value,
  ) async {
    final prefs = await SharedPreferences.getInstance();

    await prefs.setString(
      'domino_background',
      value,
    );

    if (!mounted) return;

    setState(() {
      selectedBackground = value;
    });
  }

  // ============================================================
  // EDITAR META
  // ============================================================

  void _editarMetaPuntosDialog() {
    if (_isGameOver) return;

      text: widget.match.maxPoints.toString(),

      context: context,

          'Meta de puntos a competir',

            fontWeight: FontWeight.bold,

          ),
        ),
        content: TextField(
          controller: controller,
          autofocus: true,
          keyboardType: TextInputType.number,
          inputFormatters: [
            FilteringTextInputFormatter.digitsOnly,
          ],

            labelText: 'Nueva meta',
            suffixText: 'pts',
            border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
            ),
          ),

          TextButton(
            onPressed: () => Navigator.pop(context),

                  int.tryParse(controller.text) ?? widget.match.maxPoints;

                setState(() {
                  widget.match.maxPoints = nuevaMeta;
                });
              }
              _guardarMesaActivaAuto();

              Navigator.pop(context);

              _comprobarGanadorSilencioso();
            },
            child: const Text('Guardar'),
          ),
        ],
      ),
    );
  }

  // ============================================================
  // CORREGIR / ELIMINAR ANOTACIÓN
  // ============================================================

  void _corregirOEliminarAnotacionDialog(
    int playerIndex,
    int historyIndex,
  ) {
    if (_isGameOver) return;

    final valorActual = widget.match.history[playerIndex][historyIndex];

    final controller = TextEditingController(
      text: valorActual.toString(),
    );

    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        backgroundColor: const Color(0xFFFFF9F0),
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
        ),
        title: Text(
          'Anotación de ${widget.match.players[playerIndex]}',
          style: const TextStyle(
            fontWeight: FontWeight.bold,
            color: Color(0xFF432612),
          ),
        ),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text(
              'Si te equivocaste digitando esta mano, puedes corregir su valor o eliminarla por completo:',
              style: TextStyle(
                fontSize: 13,
                color: Colors.grey,
              ),
            ),
            const SizedBox(height: 16),
            TextField(
              controller: controller,
              autofocus: true,
              keyboardType: TextInputType.number,
              inputFormatters: [
                FilteringTextInputFormatter.digitsOnly,
              ],
              decoration: InputDecoration(
                labelText: 'Puntos corregidos',
                suffixText: 'pts',
                border: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(12),
                ),
              ),
            ),
          ],
        ),
        actions: [
          TextButton.icon(
            icon: const Icon(
              Icons.delete_forever,
              color: Colors.redAccent,
            ),
            label: const Text(
              'Eliminar mano',
              style: TextStyle(
                color: Colors.redAccent,
              ),
            ),
            onPressed: () {
              setState(() {
                final eliminado =
                    widget.match.history[playerIndex].removeAt(historyIndex);

                widget.match.scores[playerIndex] -= eliminado;
              });
              _guardarMesaActivaAuto();

              Navigator.pop(context);
            },
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFF5D3218),
              foregroundColor: Colors.white,
            ),
            onPressed: () {
              final nuevoValor =
                  int.tryParse(controller.text) ?? valorActual;

              final diferencia = nuevoValor - valorActual;

              setState(() {
                widget.match.history[playerIndex][historyIndex] = nuevoValor;

                widget.match.scores[playerIndex] += diferencia;
              });

              _guardarMesaActivaAuto();

              Navigator.pop(context);

              _comprobarGanadorSilencioso();
            },
            child: const Text('Guardar'),
          ),
        ],
      ),
    );
  }

  void _reiniciarPartida() {
    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        backgroundColor: const Color(0xFFFFF9F0),
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
        ),
        title: const Text(
          'Reiniciar Partida',
          style: TextStyle(
            color: Color(0xFF432612),
            fontWeight: FontWeight.bold,
          ),
        ),
        content: const Text(
          '¿Estás seguro de que deseas reiniciar los puntos de todos los jugadores a 0?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Cancelar'),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFF5D3218),
              foregroundColor: Colors.white,
            ),
            onPressed: () {
              setState(() {
                _isGameOver = false;

                //widget.match.isFinished = false;
                widget.match.winnerName = null;

                  widget.match.history[i].clear();
                }
              });
              _limpiarMesaActivaGuardada();

              Navigator.pop(context);
            },
            child: const Text('Reiniciar'),

  // ============================================================
  // EDITAR JUGADORES
  // ============================================================

  void _editarJugadoresDialog() {
    if (_isGameOver) return;

    final controllers = widget.match.players
        .map(
          (p) => TextEditingController(text: p),
        )
        .toList();

    showDialog(
      context: context,
      builder: (context) {
        return StatefulBuilder(
          builder: (context, setDialogState) {
            return AlertDialog(
              backgroundColor: const Color(0xFFFFF9F0),
              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.circular(20),
              ),
              title: const Text(
                'Editar Mesa',
                style: TextStyle(
                  fontWeight: FontWeight.bold,
                  color: Color(0xFF432612),
                ),
              ),
              content: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    ...List.generate(
                      controllers.length,
                      (index) {
                        return Padding(
                          padding: const EdgeInsets.only(
                            bottom: 10.0,
                          ),
                          child: Row(
                            children: [
                              Expanded(
                                child: TextField(
                                  controller: controllers[index],
                                  decoration: InputDecoration(
                                    labelText: 'Participante ${index + 1}',
                                    filled: true,
                                    fillColor: Colors.white,
                                    contentPadding: const EdgeInsets.symmetric(
                                      horizontal: 12,
                                      vertical: 8,
                                    ),
                                    border: OutlineInputBorder(
                                      borderRadius: BorderRadius.circular(
                                        12,
                                      ),
                                    ),
                                  ),
                                ),
                              ),
                              if (controllers.length > 2)
                                IconButton(
                                  icon: const Icon(
                                    Icons.remove_circle_outline,
                                    color: Colors.redAccent,
                                  ),
                                  onPressed: () {
                                    setDialogState(() {
                                      controllers.removeAt(
                                        index,
                                      );
                                    });
                                  },
                                ),
                            ],
                          ),
                        );
                      },
                    ),
                    if (controllers.length < 4)
                      TextButton.icon(
                        icon: const Icon(
                          Icons.person_add_alt_1,
                          color: Color(0xFF5D3218),
                        ),
                        label: const Text(
                          'Agregar Jugador',
                          style: TextStyle(
                            color: Color(0xFF5D3218),
                            fontWeight: FontWeight.bold,
                          ),
                        ),
                        onPressed: () {
                          setDialogState(() {
                            controllers.add(
                              TextEditingController(
                                text: 'Jugador ${controllers.length + 1}',
                              ),
                            );
                          });
                        },
                      ),
                  ],
                ),
              ),
              actions: [
                TextButton(
                  onPressed: () => Navigator.pop(context),
                  child: const Text('Cancelar'),
                ),
                ElevatedButton(
                  style: ElevatedButton.styleFrom(
                    backgroundColor: const Color(0xFF5D3218),
                    foregroundColor: Colors.white,
                  ),
                  onPressed: () {
                    final nuevosNombres = controllers
                        .map(
                          (c) => c.text.trim().isEmpty
                              ? 'Jugador'
                              : c.text.trim(),
                        )
                        .toList();

                    setState(() {
                      widget.match.players = List.from(
                        nuevosNombres,
                      );

                      List<int> newScores = List.from(
                        widget.match.scores,
                      );

                      List<List<int>> newHistory = List.from(
                        widget.match.history,
                      );

                      while (newScores.length < nuevosNombres.length) {
                        newScores.add(0);
                        newHistory.add([]);
                      }

                      while (newScores.length > nuevosNombres.length) {
                        newScores.removeLast();
                        newHistory.removeLast();
                      }

                      widget.match.scores = newScores;

                      widget.match.history = newHistory;
                    });

                    _guardarMesaActivaAuto();

                    Navigator.pop(context);

                    _comprobarGanadorSilencioso();
                  },
                  child: const Text('Guardar'),
                ),
              ],
            );
          },
        );
      },
    );
  }

    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        backgroundColor: const Color(0xFFFFF9F0),
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
        ),
        title: Text(
          'Añadir puntos a ${widget.match.players[index]}',
          style: const TextStyle(
            fontWeight: FontWeight.bold,
            color: Color(0xFF432612),
          ),
        ),
        content: TextField(
          controller: controller,
          autofocus: true,
          keyboardType: TextInputType.number,
          inputFormatters: [
            FilteringTextInputFormatter.digitsOnly,
          ],
          decoration: InputDecoration(
            labelText: 'Puntos de esta mano',
            suffixText: 'pts',
            border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
            ),
          ),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Cancelar'),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFF5D3218),
              foregroundColor: Colors.white,
            ),
            onPressed: () {
              final pts = int.tryParse(controller.text) ?? 0;

              if (pts > 0) {
                setState(() {
                  widget.match.scores[index] += pts;

                  widget.match.history[index].add(pts);
                });
                _guardarMesaActivaAuto();
              }

              Navigator.pop(context);

              _comprobarGanadorSilencioso();
            },
            child: const Text('Sumar'),
          ),
        ],
      ),
    );
  }

  // ============================================================
  // COMPROBAR GANADOR
  // ============================================================

  Future<void> _comprobarGanadorSilencioso() async {
    if (_isGameOver) return;

    for (int i = 0; i < widget.match.scores.length; i++) {
      if (widget.match.scores[i] >= widget.match.maxPoints) {
        final winnerName = widget.match.players[i];

        final winnerPoints = widget.match.scores[i];

        final winnerDate = DateTime.now();

        //widget.match.isFinished = true;
        widget.match.winnerName = winnerName;
        widget.match.winnerPoints = winnerPoints;
        widget.match.winnerDate = winnerDate;

        await _limpiarMesaActivaGuardada();

        if (mounted) {
          setState(() {
            _isGameOver = true;
          });
        }

        await _guardarNuevoGanador(
  ScoreWinnerRecord(
    mesaId: widget.match.id,
    winnerName: winnerName,
    points: winnerPoints,
    date: winnerDate,
  ),
);

// Mostrar anuncio SOLO si la partida duró 30 minutos o más


if (!mounted) return;

        await showDialog(
          context: context,
          barrierDismissible: false,
          builder: (context) => Dialog(
            backgroundColor: const Color(0xFFFFF9F0),
            shape: RoundedRectangleBorder(
              borderRadius: BorderRadius.circular(28),
            ),
            child: Padding(
              padding: const EdgeInsets.symmetric(
                horizontal: 24,
                vertical: 30,
              ),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  CircleAvatar(
                    radius: 45,

                      Icons.emoji_events,

                      color: Colors.amber.shade700,

                  ),
                  const SizedBox(height: 20),
                  const Text(
                    '¡PARTIDA TERMINADA!',
                    style: TextStyle(
                      fontSize: 22,
                      fontWeight: FontWeight.w900,
                      color: Color(0xFF432612),
                    ),
                  ),
                  const SizedBox(height: 12),
                  const Text(
                    'Ganador',
                    style: TextStyle(
                      fontSize: 16,
                      color: Colors.grey,
                    ),
                  ),
                  const SizedBox(height: 4),
                  Text(
                    winnerName,
                    style: const TextStyle(
                      fontSize: 26,
                      fontWeight: FontWeight.bold,
                      color: Color(0xFF432612),
                    ),
                  ),
                  const SizedBox(height: 6),
                  Text(
                    '$winnerPoints puntos',
                    style: const TextStyle(
                      fontSize: 16,
                      fontWeight: FontWeight.w600,
                      color: Color(0xFF5D3218),
                    ),
                  ),
                  const SizedBox(height: 8),
                  Text(
                    widget.match.id,
                    style: const TextStyle(
                      fontSize: 13,
                      fontWeight: FontWeight.w700,
                      color: Colors.grey,
                    ),
                  ),
                  const SizedBox(height: 28),
                  SizedBox(
                    width: double.infinity,
                    height: 48,
                    child: ElevatedButton(
                      style: ElevatedButton.styleFrom(
                        backgroundColor: const Color(0xFF5D3218),
                        foregroundColor: Colors.white,
                        shape: RoundedRectangleBorder(
                          borderRadius: BorderRadius.circular(
                            16,
                          ),
                        ),
                        elevation: 2,
                      ),
                      onPressed: () async {
                      Navigator.pop(context);

                      await Future.delayed(
                      const Duration(milliseconds: 300),
                       );

                       _showInterstitialIfNeeded();

                       widget.onGameFinished();
                      },

                        style: TextStyle(

                      ),
                    ),

          ),

      }
