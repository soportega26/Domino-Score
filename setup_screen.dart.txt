  // CARGAR DATOS

  Future<void> _cargarDatos() async {
    final prefs = await SharedPreferences.getInstance();

    if (!mounted) return;

    final historialJson = prefs.getString(_prefHistorialKey);

    List<ScoreWinnerRecord> historial = [];

    if (historialJson != null && historialJson.isNotEmpty) {
      try {
        final List<dynamic> decoded = jsonDecode(historialJson);

        historial = decoded
            .whereType<Map<String, dynamic>>()
            .map(
              (item) => ScoreWinnerRecord.fromJson(item),
            )
            .toList();
      } catch (e) {
        debugPrint('Error cargando historial: $e');
      }
    }

    MatchModel? mesa1Guardada;
  final String? jsonMesa1 = prefs.getString('domino_mesa_activa_Mesa 1');
  if (jsonMesa1 != null && jsonMesa1.isNotEmpty) {
    try {
      final Map<String, dynamic> mapa = jsonDecode(jsonMesa1);
      final matchTemp = MatchModel.fromJson(mapa);
      if (!matchTemp.estaFinalizado) {
        mesa1Guardada = matchTemp;
      }
    } catch (e) {
      debugPrint('Error al deserializar la Mesa 1: $e');
    }
  }

  // 🔹 Cargar Mesa 2 de forma independiente
  MatchModel? mesa2Guardada;
  final String? jsonMesa2 = prefs.getString('domino_mesa_activa_Mesa 2');
  if (jsonMesa2 != null && jsonMesa2.isNotEmpty) {
    try {
      final Map<String, dynamic> mapa = jsonDecode(jsonMesa2);
      final matchTemp = MatchModel.fromJson(mapa);
      if (!matchTemp.estaFinalizado) {
        mesa2Guardada = matchTemp;
      }
    } catch (e) {
      debugPrint('Error al deserializar la Mesa 2: $e');
    }
  }

  setState(() {
      recordsGanadores = historial;
      selectedBackground =
          prefs.getString('domino_background') ?? 'dominicano';

      // 🔹 Asignación directa de las mesas cargadas previamente
      mesa1 = mesa1Guardada;
      mesa2 = mesa2Guardada;
    });

    _restaurarGanadoresDeMesas();
  }

  void _restaurarGanadoresDeMesas() {
    if (mesa1 != null) {
      final ganador = _ultimoGanadorDeMesa('Mesa 1');

      if (ganador != null) {
        mesa1!.winnerName = ganador.winnerName;
        mesa1!.winnerPoints = ganador.points;
        mesa1!.winnerDate = ganador.date;
      }
    }

    if (mesa2 != null) {
      final ganador = _ultimoGanadorDeMesa('Mesa 2');

      if (ganador != null) {
        mesa2!.winnerName = ganador.winnerName;
        mesa2!.winnerPoints = ganador.points;
        mesa2!.winnerDate = ganador.date;
      }
    }
  }

  ScoreWinnerRecord? _ultimoGanadorDeMesa(String mesaId) {
    for (final record in recordsGanadores) {
      if (record.mesaId == mesaId) {
        return record;
      }
    }

    return null;
  }

  void _confirmarNuevaPartida(String mesaId) {
    showDialog(
      context: context,
      builder: (dialogContext) => AlertDialog(
        backgroundColor: const Color(0xFFFFF9F0),
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(20),
        ),
        title: const Row(
          children: [
            Icon(Icons.warning_amber_rounded, color: Colors.orange),
            SizedBox(width: 8),
            Text('Nueva partida'),
          ],
        ),
        content: Text(
          'La partida actual de la $mesaId será reemplazada por una nueva.\n\n'
          'El historial de campeones no se eliminará.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext),
            child: const Text('Cancelar'),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFF5D3218),
              foregroundColor: Colors.white,
            ),
            onPressed: () async {
              Navigator.pop(dialogContext);
              final prefs = await SharedPreferences.getInstance();
            await prefs.remove('domino_mesa_activa_$mesaId');

            setState(() {
              if (mesaId == 'Mesa 1') mesa1 = null;
              if (mesaId == 'Mesa 2') mesa2 = null;
            });
              _configurarMesaDialog(mesaId, esNueva: true);
            },
            child: const Text('Crear nueva partida'),
          ),
        ],
      ),
    );
  }

  // ============================================================
  // CONFIGURAR MESA
  // ============================================================


void _configurarMesaDialog(String mesaId, {bool esNueva = false}) {
  final matchActual = esNueva ? null : (mesaId == 'Mesa 1' ? mesa1 : mesa2);

  showDialog(
    context: context,
    builder: (context) {
      return _ConfigurarMesaDialogContent(
        mesaId: mesaId,
        matchActual: matchActual,
        onConfirm: (listaJugadores, metaFinal) {
          
          // ✅ CÓDIGO CORREGIDO: Inicializamos winnerName, winnerPoints y winnerDate en null
          final nuevoJuego = MatchModel(
            id: mesaId,
            players: listaJugadores,
            scores: List.filled(listaJugadores.length, 0),
            history: List.generate(listaJugadores.length, (_) => []), // Limpiamos historial de anotaciones
            maxPoints: metaFinal,
            winnerName: null,   // Al ser null, la pantalla de anotación estará 100% activa
            winnerPoints: null,
            winnerDate: null,
          );

          setState(() {
            if (mesaId == 'Mesa 1') {
              mesa1 = nuevoJuego;
            } else {
              mesa2 = nuevoJuego;
            }
          });

          _irAlMarcador(nuevoJuego);
        },
      );
    },
  );
}

  // ============================================================
  // ABRIR MARCADOR
  // ============================================================

  void _irAlMarcador(MatchModel match) {
    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => ScoreScreen(
          match: match,
          onGameFinished: () async {
            await _cargarDatos();
          },
        ),
      ),
    ).then((_) async {
      if (!mounted) return;

      await _cargarDatos();

      setState(() {});
    });
  }

class _ConfigurarMesaDialogContent extends StatefulWidget {
  final String mesaId;
  final MatchModel? matchActual;

  final Function(
    List<String> players,
    int maxPoints,
  ) onConfirm;

  const _ConfigurarMesaDialogContent({
    required this.mesaId,
    required this.matchActual,
    required this.onConfirm,
  });

  @override
  State<_ConfigurarMesaDialogContent> createState() =>
      __ConfigurarMesaDialogContentState();
}

class __ConfigurarMesaDialogContentState
    extends State<_ConfigurarMesaDialogContent> {
  final List<TextEditingController> _participantControllers = [];

  late TextEditingController puntosCustomController;

  int puntosMesa = 200;

  @override
  void initState() {
    super.initState();

    final match = widget.matchActual;

    if (match != null && match.players.isNotEmpty) {
      for (final player in match.players) {
        _participantControllers.add(
          TextEditingController(text: player),
        );
      }
    } else {
      _participantControllers.add(
        TextEditingController(text: 'Jugador 1'),
      );

      _participantControllers.add(
        TextEditingController(text: 'Jugador 2'),
      );
    }

    puntosMesa = match?.maxPoints ?? 200;

    puntosCustomController = TextEditingController(
      text: puntosMesa.toString(),
    );
  }

  void _addParticipant() {
    if (_participantControllers.length >= 4) {
      return;
    }

    setState(() {
      _participantControllers.add(
        TextEditingController(
          text: 'Jugador ${_participantControllers.length + 1}',
        ),
      );
    });
  }

  void _removeParticipant(int index) {
    if (_participantControllers.length <= 2) {
      return;
    }

    setState(() {
      final controller = _participantControllers.removeAt(index);
      controller.dispose();
    });
  }

  @override
  void dispose() {
    for (final controller in _participantControllers) {
      controller.dispose();
    }

    puntosCustomController.dispose();

    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final isEditing = widget.matchActual != null;

    return AlertDialog(
      backgroundColor: const Color(0xFFFFF9F0),
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(25),
      ),
      title: Row(
        children: [
          Icon(
            isEditing ? Icons.edit_note : Icons.settings,
            color: const Color(0xFF5D3218),
          ),
          const SizedBox(width: 10),
          Text(
            isEditing ? 'Editar ${widget.mesaId}' : 'Configurar ${widget.mesaId}',
            style: const TextStyle(
              fontWeight: FontWeight.w900,
              color: Color(0xFF432612),
            ),
          ),
        ],
      ),
      content: SingleChildScrollView(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              mainAxisAlignment: MainAxisAlignment.spaceBetween,
              children: [
                const Text(
                  'Participantes (2-4):',
                  style: TextStyle(
                    fontWeight: FontWeight.w900,
                    color: Color(0xFF432612),
                  ),
                ),
                if (_participantControllers.length < 4)
                  TextButton.icon(
                    onPressed: _addParticipant,
                    icon: const Icon(Icons.add, size: 18),
                    label: const Text('Agregar'),
                    style: TextButton.styleFrom(
                      foregroundColor: const Color(0xFF5D3218),
                    ),
                  ),
              ],
            ),
            const SizedBox(height: 8),
            ...List.generate(
              _participantControllers.length,
              (index) {
                return Padding(
                  padding: const EdgeInsets.only(bottom: 10),
                  child: Row(
                    children: [
                      Expanded(
                        child: TextField(
                          controller: _participantControllers[index],
                          textCapitalization: TextCapitalization.words,
                          decoration: InputDecoration(
                            labelText: 'Jugador ${index + 1}',
                            prefixIcon: const Icon(Icons.person_outline),
                            isDense: true,
                            border: OutlineInputBorder(
                              borderRadius: BorderRadius.circular(12),
                            ),
                          ),
                        ),
                      ),
                      if (_participantControllers.length > 2)
                        IconButton(
                          icon: const Icon(
                            Icons.remove_circle_outline,
                            color: Colors.redAccent,
                          ),
                          onPressed: () => _removeParticipant(index),
                          tooltip: 'Quitar jugador',
                        ),
                    ],
                  ),
                );
              },
            ),
            const SizedBox(height: 16),
            const Text(
              'Meta de puntos:',
              style: TextStyle(
                fontWeight: FontWeight.w900,
                color: Color(0xFF432612),
              ),
            ),
            const SizedBox(height: 10),
            Row(
              children: [100, 200, 300].map(
                (points) {
                  final selected = puntosMesa == points;

                  return Expanded(
                    child: Padding(
                      padding: const EdgeInsets.symmetric(horizontal: 2),
                      child: ChoiceChip(
                        selected: selected,
                        label: Text('$points'),
                        onSelected: (_) {
                          setState(() {
                            puntosMesa = points;
                            puntosCustomController.text = points.toString();
                          });
                        },
                        selectedColor: const Color(0xFF5D3218),
                        labelStyle: TextStyle(
                          color: selected ? Colors.white : const Color(0xFF5D3218),
                          fontWeight: FontWeight.w800,
                        ),
                      ),
                    ),
                  );
                },
              ).toList(),
            ),
            const SizedBox(height: 12),
            TextField(
              controller: puntosCustomController,
              keyboardType: TextInputType.number,
              inputFormatters: [
                FilteringTextInputFormatter.digitsOnly,
              ],
              onChanged: (val) {
                final parsed = int.tryParse(val);
                if (parsed != null && parsed > 0) {
                  setState(() {
                    puntosMesa = parsed;
                  });
                } else {
                  setState(() {
                    puntosMesa = -1;
                  });
                }
              },
              decoration: InputDecoration(
                labelText: 'Puntos personalizados',
                prefixIcon: const Icon(Icons.flag_outlined),
                suffixText: 'pts',
                isDense: true,
                border: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(12),
                ),
              ),
            ),
          ],
        ),
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.pop(context),
          child: const Text('Cancelar'),
        ),
        ElevatedButton.icon(
          icon: Icon(isEditing ? Icons.save : Icons.play_arrow),
          label: Text(isEditing ? 'Guardar' : 'Comenzar'),
          style: ElevatedButton.styleFrom(
            backgroundColor: const Color(0xFF5D3218),
            foregroundColor: Colors.white,
          ),
          onPressed: () {
            List<String> names = _participantControllers
                .map((c) => c.text.trim())
                .where((t) => t.isNotEmpty)
                .toList();

            if (names.length < 2) {
              names = ['Jugador 1', 'Jugador 2'];
            }

            final metaParsed = int.tryParse(puntosCustomController.text.trim());

            final metaFinal = (metaParsed != null && metaParsed > 0)
                ? metaParsed
                : (puntosMesa > 0 ? puntosMesa : 200);

            Navigator.pop(context);

            widget.onConfirm(
              names,
              metaFinal,
            );
          },
        ),
      ],
    );
  }
}
